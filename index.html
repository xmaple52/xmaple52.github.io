Weapon.cs C#

Хитскан с разбросом из шутера: попадание считается одним лучом, без пуль-объектов.

// hitscan с конусом разброса — прототип оружия
public bool Fire(Vector3 origin, Vector3 forward, out RaycastHit hit)
{
    var spread = Quaternion.Euler(
        Random.Range(-kick.x, kick.x),
        Random.Range(-kick.y, kick.y),
        0f);

    var dir = spread * forward;

    return Physics.Raycast(
        origin, dir, out hit, range,
        hitMask, QueryTriggerInteraction.Ignore);
}
